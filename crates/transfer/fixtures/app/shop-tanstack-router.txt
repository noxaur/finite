=== README.md
# Shop

A clickable prototype exported from Finite. It uses Vite, React and tanstack-router.

```sh
npm install
npm run dev      # http://localhost:5173
npm run build    # writes dist/
```

Each route is a file in `src/pages/` that holds the route's screens (one per artboard width) and shows the one that fits the window. The first route is served at `/`. Images are in `public/assets/`, and the design's tokens are CSS variables in `src/global.css`.

=== src/router.tsx
import { RouterProvider, createRootRoute, createRoute, createRouter } from "@tanstack/react-router";
import PageHome from "./pages/PageHome";
import PageCart from "./pages/PageCart";

const rootRoute = createRootRoute();
const route0 = createRoute({ getParentRoute: () => rootRoute, path: "/", component: PageHome });
const route1 = createRoute({ getParentRoute: () => rootRoute, path: "/cart", component: PageCart });

const router = createRouter({
  routeTree: rootRoute.addChildren([route0, route1]),
  caseSensitive: true,
  scrollRestoration: true,
});

declare module "@tanstack/react-router" {
  interface Register {
    router: typeof router;
  }
}

export function App() {
  return <RouterProvider router={router} />;
}
